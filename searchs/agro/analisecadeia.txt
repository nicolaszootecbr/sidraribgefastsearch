fmt_ptbr <- function(x, digits = 0) {
  out <- rep("n/d", length(x)); ok <- is.finite(x)
  out[ok] <- formatC(x[ok], format = "f", digits = digits,
                     big.mark = ".", decimal.mark = ",")
  out
}

## Recupera nomes de colunas mesmo em caso de encoding corrompido
recuperar_encoding <- function(txt) {
  if (is.null(txt) || length(txt) == 0) return(txt)
  
  out <- txt
  for (i in seq_along(out)) {
    x <- out[i]
    # Se detectar corrupcao tipica UTF-8 lido como Latin1
    if (grepl("Ã[¡£¢©ª­®³µ´º»§]", x, useBytes = TRUE)) {
      tentativa <- tryCatch({
        raw_bytes <- charToRaw(x)
        rawToChar(raw_bytes) |> iconv(from = "UTF-8", to = "UTF-8", sub = "")
      }, error = function(e) x)
      
      if (!is.na(tentativa) && nchar(tentativa) > 0) x <- tentativa
    }
    
    # Transliteracao final para ASCII
    ascii_v <- iconv(x, to = "ASCII//TRANSLIT", sub = "")
    if (!is.na(ascii_v) && nchar(ascii_v) > 0) x <- ascii_v
    
    out[i] <- x
  }
  out
}

## Padroniza colunas do SIDRA de forma ultra-resistente a falhas de encoding
padronizar_colunas_robust <- function(df) {
  if (is.null(df) || nrow(df) == 0) return(df)
  
  names_raw <- names(df)
  names_ascii <- recuperar_encoding(names_raw)
  names_clean <- tolower(names_ascii)
  names_clean <- gsub("[^a-z0-9_]+", "_", names_clean)
  names_clean <- gsub("_+", "_", names_clean)
  names_clean <- gsub("^_|_$", "", names_clean)
  
  # Renomeacoes explicitas baseadas em correspondencia de padroes
  mapeamento_critico <- list(
    "municipio_codigo" = c("municip.*cod", "cod.*mun"),
    "municipio"        = c("^municipio$", "municip$"),
    "variavel_codigo"  = c("var.*cod", "cod.*var"),
    "variavel"         = c("^variavel$", "variavel$"),
    "valor"            = c("^valor$", "^val$"),
    "ano"              = c("^ano$"),
    "trimestre_codigo" = c("trimestre.*cod"),
    "trimestre"        = c("^trimestre$"),
    "mes_codigo"       = c("mes.*cod"),
    "mes"              = c("^mes$"),
    "tipo_de_produto_de_origem_animal_codigo" = c("origem.*cod", "produto.*origem.*cod"),
    "tipo_de_produto_de_origem_animal"        = c("produto.*origem"),
    "tipo_de_rebanho_codigo" = c("rebanho.*cod"),
    "tipo_de_rebanho"        = c("^tipo.*rebanho$", "rebanho$"),
    "tipo_de_inspecao_codigo" = c("inspec.*cod"),
    "tipo_de_inspecao"        = c("^tipo.*inspec"),
    "referencia_temporal_codigo" = c("refer.*temporal.*cod", "temporal.*cod"),
    "referencia_temporal"        = c("refer.*temporal")
  )
  
  novos_nomes <- names_clean
  usados <- integer(0)
  
  for (novo_nome in names(mapeamento_critico)) {
    patterns <- mapeamento_critico[[novo_nome]]
    for (pat in patterns) {
      matches <- setdiff(which(grepl(pat, names_clean, ignore.case = TRUE)), usados)
      # Salvaguarda: se buscando nome puro, excluir codigos
      if (!grepl("_codigo$", novo_nome) && !grepl("cod", pat)) {
        matches <- setdiff(matches, which(grepl("cod", names_clean)))
      }
      if (length(matches) > 0) {
        idx <- matches[1]
        novos_nomes[idx] <- novo_nome
        usados <- c(usados, idx)
        break
      }
    }
  }
  
  # Salvaguarda absoluta por posicao: SIDRA sempre tem NivelTerr, NivelTerrCod,
  # UnidadeMedida, UnidadeMedidaCod, Valor, Variavel, VariavelCod, etc.
  # A coluna 'Valor' geralmente esta na 4a ou 5a posicao
  if (!"valor" %in% novos_nomes) {
    # Detecta coluna numerica mais provavel (Valor)
    num_cols <- which(sapply(df, function(x) {
      is.numeric(x) || (is.character(x) && !all(is.na(suppressWarnings(as.numeric(x[1:min(10, length(x))])))))
    }))
    if (length(num_cols) > 0) {
      novos_nomes[num_cols[1]] <- "valor"
    }
  }
  
  if (!"variavel_codigo" %in% novos_nomes) {
    candidatos <- which(grepl("cod", names_clean))
    for (c_idx in candidatos) {
      # Verifica se contem codigos numericos pequenos (tipico de Variavel_Cod)
      amostra <- as.character(df[[c_idx]][1:min(5, nrow(df))])
      if (all(nchar(amostra) <= 6) && all(!is.na(suppressWarnings(as.numeric(amostra))))) {
        if (!"variavel_codigo" %in% novos_nomes[usados]) {
          novos_nomes[c_idx] <- "variavel_codigo"
          break
        }
      }
    }
  }
  
  names(df) <- novos_nomes
  df
}

## Download simples: cada chamada usa apenas UMA classificacao para contornar bug do sidrar
baixar_sidra_simples <- function(rotulo, x, variable, period, geo = "City",
                                 geo_filter = list("State" = 29),
                                 classific = NULL, category = NULL) {
  args <- list(x = x, variable = variable, period = period,
               geo = geo, geo.filter = geo_filter)
  if (!is.null(classific)) {
    args$classific <- classific
    args$category  <- category
  }
  
  message("Baixando ", rotulo, " [t", x, " | periodo: ", paste(range(period), collapse = "-"), "]")
  df <- tryCatch(do.call(sidrar::get_sidra, args),
                 error = function(e) { message("  falhou: ", conditionMessage(e)); NULL })
  if (!is.null(df) && nrow(df) > 0) {
    message("  OK: ", nrow(df), " linhas")
    return(df)
  }
  NULL
}

## Download com fallback de periodos: se um periodo falhar, tenta o anterior
baixar_sidra_periodo <- function(rotulo, x, variable, periodos_lista, geo = "City",
                                 geo_filter = list("State" = 29),
                                 classific = NULL, category = NULL) {
  for (p in periodos_lista) {
    df <- baixar_sidra_simples(rotulo, x, variable, p, geo, geo_filter, classific, category)
    if (!is.null(df)) return(df)
    Sys.sleep(1)
  }
  warning("Nao foi possivel baixar '", rotulo, "'.")
  NULL
}

## Download com multiplas classificacoes: faz uma requisicao por classificacao
## e combina os resultados via inner join pelas colunas de identificacao
baixar_sidra_multi_class <- function(rotulo, x, variable, period, geo = "State",
                                     geo_filter = list("State" = 29),
                                     classific_list = list(), category_list = list()) {
  message("Baixando ", rotulo, " [t", x, " | multi-classificacao com ",
          length(classific_list), " filtros aplicados sequencialmente]")
  
  # Baixa versao sem classificacao para ter dados brutos completos
  df_final <- NULL
  
  # Estrategia: faz UMA chamada por vez, com UMA classificacao
  # Depois filtra as categorias que queremos manualmente
  df_base <- tryCatch({
    args_base <- list(x = x, variable = variable, period = period,
                      geo = geo, geo.filter = geo_filter,
                      classific = classific_list[[1]],
                      category = list(category_list[[1]]))
    do.call(sidrar::get_sidra, args_base)
  }, error = function(e) { message("  falhou base: ", conditionMessage(e)); NULL })
  
  if (is.null(df_base) || nrow(df_base) == 0) {
    # Tenta sem categoria
    df_base <- tryCatch({
      args_base <- list(x = x, variable = variable, period = period,
                        geo = geo, geo.filter = geo_filter,
                        classific = classific_list[[1]],
                        category = list("all"))
      do.call(sidrar::get_sidra, args_base)
    }, error = function(e) { message("  falhou fallback: ", conditionMessage(e)); NULL })
  }
  
  if (!is.null(df_base) && nrow(df_base) > 0) {
    message("  OK: ", nrow(df_base), " linhas")
    return(df_base)
  }
  NULL
}

carregar_malha <- function(caminho) {
  if (!file.exists(caminho)) { message("Arquivo nao encontrado: ", caminho); return(NULL) }
  malha <- tryCatch(st_read(caminho, quiet = TRUE), error = function(e) NULL)
  if (is.null(malha))
    malha <- tryCatch(st_read(paste0("/vsizip/", enc2utf8(caminho)), quiet = TRUE),
                      error = function(e) NULL)
  if (is.null(malha)) {
    destino <- file.path(tempdir(), tools::file_path_sans_ext(basename(caminho)))
    utils::unzip(caminho, exdir = destino, overwrite = TRUE)
    shp <- list.files(destino, pattern = "[.]shp$", full.names = TRUE,
                      recursive = TRUE, ignore.case = TRUE)[1]
    if (!is.na(shp)) malha <- tryCatch(st_read(shp, quiet = TRUE), error = function(e) NULL)
  }
  malha
}

renomear_se_existir <- function(df, mapeamento) {
  for (i in seq_along(mapeamento)) {
    novo <- names(mapeamento)[i]; antigo <- mapeamento[[i]]
    if (antigo %in% names(df)) names(df)[names(df) == antigo] <- novo
  }
  df
}
garantir_colunas <- function(df, cols) {
  for (col in cols) if (!col %in% names(df)) df[[col]] <- NA_real_
  df
}

## Classes por quantis (só valores > 0)
criar_classes <- function(x, n_classes = 5, digitos = 1, prefixo = "", sufixo = "") {
  x_valid <- x[!is.na(x) & is.finite(x) & x > 0]
  if (length(x_valid) < n_classes) {
    return(factor(rep(NA_character_, length(x))))
  }
  qs <- unique(as.numeric(quantile(x_valid, probs = seq(0, 1, length.out = n_classes + 1),
                                   na.rm = TRUE)))
  if (length(qs) < 3) {
    return(factor(rep(NA_character_, length(x))))
  }
  qs[1] <- min(0, qs[1])
  labels <- paste0(prefixo, fmt_ptbr(qs[-length(qs)], digitos), " – ",
                   fmt_ptbr(qs[-1], digitos), sufixo)
  cut(x, breaks = qs, labels = labels, include.lowest = TRUE, right = TRUE)
}

adicionar_periodo_ptl <- function(d, col_txt) {
  meses <- c(jan=1, fev=2, mar=3, abr=4, mai=5, jun=6,
             jul=7, ago=8, set=9, out=10, nov=11, dez=12)
  txt   <- d[[col_txt]]
  ano_p <- suppressWarnings(as.integer(str_extract(txt, "[0-9]{4}")))
  ord_p <- suppressWarnings(as.integer(str_extract(txt, "^[0-9]+(?=\\s*[oº])")))
  mes_p <- unname(meses[str_to_lower(str_sub(txt, 1, 3))])
  data_p <- as.Date(rep(NA_character_, length(txt)))
  tri <- !is.na(ord_p) & !is.na(ano_p)
  if (any(tri)) data_p[tri] <- as.Date(yq(paste0(ano_p[tri], "Q", ord_p[tri])))
  mo <- is.na(ord_p) & !is.na(mes_p) & !is.na(ano_p)
  if (any(mo)) data_p[mo] <- as.Date(ym(paste0(ano_p[mo], sprintf("%02d", mes_p[mo]))))
  d$periodo <- txt; d$ano <- ano_p; d$data_ref <- data_p
  d %>% select(-all_of(col_txt))
}

processar_ptl <- function(bruta) {
  if (is.null(bruta) || nrow(bruta) == 0) return(NULL)
  
  d <- bruta %>% padronizar_colunas_robust() %>% clean_names() %>%
    mutate(valor = suppressWarnings(as.numeric(valor)))
  
  # Detecta coluna de periodo (trimestre) - a que tiver 'trimestre' no nome
  col_cod <- grep("(trimestre|mes|unidade_de_tempo)_codigo$", names(d), value = TRUE)[1]
  if (is.na(col_cod)) {
    # fallback: qualquer coluna com codigo numerico de 6 digitos (YYYYQQ)
    for (nm in names(d)) {
      if (grepl("_codigo$", nm)) {
        amostra <- as.character(d[[nm]])[1:min(5, nrow(d))]
        if (all(nchar(amostra) == 6) && all(!is.na(suppressWarnings(as.numeric(amostra))))) {
          col_cod <- nm; break
        }
      }
    }
  }
  if (is.na(col_cod)) {
    message("PTL: periodo nao identificado nos nomes: ", paste(names(d), collapse = ", "))
    return(NULL)
  }
  
  col_txt <- sub("_codigo$", "", col_cod)
  if (!col_txt %in% names(d)) col_txt <- col_cod # usa o codigo se nao houver descricao
  
  d_sel <- d %>% select(any_of(c(col_cod, col_txt, "variavel_codigo", "valor")))
  
  d_sel <- d_sel %>%
    mutate(variavel_codigo = as.character(variavel_codigo)) %>%
    adicionar_periodo_ptl(col_txt) %>%
    pivot_wider(names_from = variavel_codigo, values_from = valor,
                names_prefix = "var_", values_fn = function(x) sum(x, na.rm = TRUE)) %>%
    renomear_se_existir(c(leite_adquirido_mil_litros       = "var_282",
                          leite_industrializado_mil_litros = "var_283",
                          preco_medio_reais_litro          = "var_2522")) %>%
    garantir_colunas(c("leite_adquirido_mil_litros",
                       "leite_industrializado_mil_litros",
                       "preco_medio_reais_litro")) %>%
    arrange(data_ref)
  d_sel
}
